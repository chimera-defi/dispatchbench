```python
def overlap_minutes(meetings: list[list[int]], k: int) -> int:
    delta = {}
    for start, end in meetings:
        if start < end:
            delta[start] = delta.get(start, 0) + 1
            delta[end] = delta.get(end, 0) - 1

    active = 0
    prev = None
    total = 0

    for pos in sorted(delta):
        if prev is not None and active >= k:
            total += pos - prev
        active += delta[pos]
        prev = pos

    return total
```
