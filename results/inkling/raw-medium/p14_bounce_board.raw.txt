from fractions import Fraction

def expected_turns(n: int, m: int, jumps: list[list[int]]) -> list[int] | None:
    jump_to = {}
    for a, b in jumps:
        jump_to[a] = b

    def settle(x):
        while x in jump_to:
            x = jump_to[x]
        return x

    end_prob = {}
    trans = {}
    for p in range(0, n):
        if p in jump_to:
            continue
        end = Fraction(0)
        tr = {}
        for r in range(1, m + 1):
            nxt = p + r
            if nxt > n:
                nxt = 2 * n - p - r
            final = settle(nxt)
            if final == n:
                end += Fraction(1, m)
            else:
                tr[final] = tr.get(final, Fraction(0)) + Fraction(1, m)
        end_prob[p] = end
        trans[p] = tr

    seen = {0}
    stack = [0]
    reachable = [0]
    while stack:
        p = stack.pop()
        for j, prob in trans[p].items():
            if prob > 0 and j not in seen:
                seen.add(j)
                stack.append(j)
                reachable.append(j)

    can_reach_end = {p: end_prob[p] > 0 for p in reachable}
    changed = True
    while changed:
        changed = False
        for p in reachable:
            if can_reach_end[p]:
                continue
            for j, prob in trans[p].items():
                if prob > 0 and can_reach_end.get(j, False):
                    can_reach_end[p] = True
                    changed = True
                    break

    if any(not can_reach_end[p] for p in reachable):
        return None

    k = len(reachable)
    idx = {p: i for i, p in enumerate(reachable)}
    mat = []
    for i, p in enumerate(reachable):
        row = [Fraction(0) for _ in range(k)]
        row[i] = Fraction(1)
        for j, prob in trans[p].items():
            row[idx[j]] -= prob
        mat.append(row + [Fraction(1)])

    for col in range(k):
        pivot = -1
        best = Fraction(0)
        for r in range(col, k):
            v = mat[r][col]
            if v != 0 and (pivot == -1 or abs(v) > best):
                pivot = r
                best = abs(v)
        if pivot == -1:
            return None
        if pivot != col:
            mat[col], mat[pivot] = mat[pivot], mat[col]
        pv = mat[col][col]
        for r in range(k):
            if r == col:
                continue
            f = mat[r][col] / pv
            if f == 0:
                continue
            for c in range(col, k + 1):
                mat[r][c] -= f * mat[col][c]

    x = [Fraction(0) for _ in range(k)]
    for i in range(k):
        if mat[i][i] == 0:
            return None
        x[i] = mat[i][k] / mat[i][i]

    ans = x[idx[0]]
    return [ans.numerator, ans.denominator]
