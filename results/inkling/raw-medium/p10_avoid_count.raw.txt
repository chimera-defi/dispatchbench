```python
def avoid_count(n: int, alphabet: str, banned: list[str]) -> int:
    MOD = 1_000_000_007

    if any(b == "" for b in banned):
        return 0

    patterns = list(set(banned))
    minimal = []
    for p in patterns:
        redundant = False
        for q in patterns:
            if q != p and q in p:
                redundant = True
                break
        if not redundant:
            minimal.append(p)

    state_set = {""}
    for p in minimal:
        for i in range(1, len(p)):
            state_set.add(p[:i])

    states = list(state_set)
    idx = {s: i for i, s in enumerate(states)}
    m = len(states)

    A = [[0] * m for _ in range(m)]
    for i, s0 in enumerate(states):
        for c in alphabet:
            cur = s0 + c
            bad = False
            for p in minimal:
                if len(p) <= len(cur) and cur.endswith(p):
                    bad = True
                    break
            if bad:
                continue

            best = ""
            L = len(cur)
            while L > 0:
                suf = cur[len(cur) - L:]
                if suf in idx:
                    best = suf
                    break
                L -= 1

            A[i][idx[best]] += 1

    def mat_mul(X, Y):
        Z = [[0] * m for _ in range(m)]
        for i in range(m):
            Xi = X[i]
            Zi = Z[i]
            for k in range(m):
                x = Xi[k]
                if x == 0:
                    continue
                Yk = Y[k]
                for j in range(m):
                    Zi[j] = (Zi[j] + x * Yk[j]) % MOD
        return Z

    def mat_pow(X, e):
        R = [[1 if i == j else 0 for j in range(m)] for i in range(m)]
        while e:
            if e & 1:
                R = mat_mul(R, X)
            e >>= 1
            if e:
                X = mat_mul(X, X)
        return R

    M = mat_pow(A, n)
    return sum(M[0]) % MOD
```
