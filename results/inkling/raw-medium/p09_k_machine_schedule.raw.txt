def k_machine_schedule(jobs: list[list[int]], k: int) -> int:
    if k <= 0 or not jobs:
        return 0
    times = sorted({t for s, e, _ in jobs for t in (s, e)})
    idx = {t: i for i, t in enumerate(times)}
    m = len(times)
    if m < 2:
        return 0

    # edges: [u, v, cap, cost, rev_index]
    edges = []

    def add_edge(u, v, cap, cost):
        edges.append([u, v, cap, cost, len(edges) + 1])
        edges.append([v, u, 0, -cost, len(edges) - 1])

    for i in range(m - 1):
        add_edge(i, i + 1, k, 0)
    for s, e, p in jobs:
        add_edge(idx[s], idx[e], 1, -p)

    source, sink = 0, m - 1
    flow = 0
    total_cost = 0

    while flow < k:
        dist = [float('inf')] * m
        dist[source] = 0
        prev_edge = [-1] * m
        for _ in range(m - 1):
            updated = False
            for ei, (u, v, cap, cost, _) in enumerate(edges):
                if cap > 0 and dist[u] + cost < dist[v]:
                    dist[v] = dist[u] + cost
                    prev_edge[v] = ei
                    updated = True
            if not updated:
                break
        if dist[sink] == float('inf'):
            break
        # reconstruct path
        path = []
        v = sink
        ok = True
        while v != source:
            ei = prev_edge[v]
            if ei == -1:
                ok = False
                break
            path.append(ei)
            v = edges[ei][0]
        if not ok or v != source:
            break
        bottleneck = k
        for ei in path:
            bottleneck = min(bottleneck, edges[ei][2])
        for ei in path:
            edges[ei][2] -= bottleneck
            edges[edges[ei][4]][2] += bottleneck
        flow += bottleneck
        total_cost += bottleneck * dist[sink]

    return -total_cost
