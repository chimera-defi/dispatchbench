        while pos < n and s[pos] == ' ':

    def trunc_div(a, b):
        if b == 0:
            raise Error("division by zero")
        q = abs(a) // abs(b)
        if (a < 0) != (b < 0):
            q = -q
        return q

        if b == 0:
            raise Error("remainder by zero")
        return a - b * trunc_div(a, b)

    def parse_primary():
        nonlocal pos
        skip()
        if pos < n and s[pos] == '(':
            pos += 1
            val = parse_additive()
            skip()
            if pos >= n or s[pos] != ')':
                raise Error("expected )")
            pos += 1
            return val

        while pos < n and '0' <= s[pos] <= '9':
            pos += 1
        if start == pos:
            raise Error("expected number")
        return int(s[start:pos])

    def parse_power():
        nonlocal pos
        skip()
        base = parse_primary()
        skip()
        if pos < n and s[pos] == '^':
            pos += 1
            exp = parse_unary()
            if exp < 0:
                raise Error("negative exponent")
            return base ** exp
        return base

    def parse_unary():
        nonlocal pos
        skip()
        if pos < n and s[pos] == '-':
            pos += 1
            return -parse_unary()
        return parse_power()

    def parse_multiplicative():
        nonlocal pos
        skip()
        val = parse_unary()
        while True:
            skip()
            if pos < n and s[pos] == '*':
                pos += 1
                val = val * parse_unary()

            elif pos < n and s[pos] == '%':

                val = val + parse_multiplicative()
            elif pos < n and s[pos] == '-':
                pos += 1
                val = val - parse_multiplicative()
            else:
                return val

    try:
        val = parse_additive()
        skip()
        if pos != n:
