import re
from typing import Any

INT_RE = re.compile(r'[0-9]+')
CELL_RE = re.compile(r'[A-Z][1-9][0-9]?')
IDENT_RE = re.compile(r'[A-Z]+')


def sheet(cells: dict[str, str]) -> dict[str, int | str]:
    memo: dict[str, int | str] = {}
    in_progress: set[str] = set()

    def is_error(v: Any) -> bool:
        return isinstance(v, str) and v.startswith('#')

    def tokenize(s: str):
        toks = []
        i = 0
        n = len(s)
        while i < n:
            c = s[i]
            if c == ' ' or c == '\t' or c == '\n' or c == '\r':
                i += 1
                continue
            if c in '0123456789':
                j = i
                while j < n and s[j].isdigit():
                    j += 1
                toks.append(('INT', s[i:j]))
                i = j
                continue
            if c == '<' and i + 1 < n and s[i + 1] == '=':
                toks.append(('OP', '<=')); i += 2; continue
            if c == '>' and i + 1 < n and s[i + 1] == '=':
                toks.append(('OP', '>=')); i += 2; continue
            if c == '<' and i + 1 < n and s[i + 1] == '>':
                toks.append(('OP', '<>')); i += 2; continue
            if c in '+-*/(),:':
                toks.append(('P', c)); i += 1; continue
            if c in '=<>':
                toks.append(('OP', c)); i += 1; continue
            if c.isalpha() and c.isascii():
                m = CELL_RE.match(s, i)
                if m:
                    toks.append(('CELL', m.group(0)))
                    i = m.end()
                    continue
                m = IDENT_RE.match(s, i)
                if m:
                    toks.append(('IDENT', m.group(0)))
                    i = m.end()
                    continue
            raise ValueError('bad char')
        toks.append(('EOF', ''))
        return toks

    def parse(toks):
        idx = 0

        def peek(k=0):
            return toks[idx + k]

        def next_():
            nonlocal idx
            t = toks[idx]
            idx += 1
            return t

        def expect_punct(p):
            t = next_()
            if not (t[0] == 'P' and t[1] == p):
                raise ValueError('expected ' + p)

        def parse_expr():
            node = parse_term()
            while True:
                t = peek()
                if t[0] == 'P' and t[1] in '+-':
                    next_()
                    op = t[1]
                    right = parse_term()
                    node = ('bin', op, node, right)
                else:
                    return node

        def parse_term():
            node = parse_unary()
            while True:
                t = peek()
                if t[0] == 'P' and t[1] in '*/':
                    next_()
                    op = t[1]
                    right = parse_unary()
                    node = ('bin', op, node, right)
                else:
                    return node

        def parse_unary():
            t = peek()
            if t[0] == 'P' and t[1] == '-':
                next_()
                return ('neg', parse_unary())
            return parse_atom()

        def parse_atom():
            t = next_()
            if t[0] == 'INT':
                return ('int', int(t[1]))
            if t[0] == 'CELL':
                return ('cell', t[1])
            if t[0] == 'P' and t[1] == '(':
                node = parse_expr()
                expect_punct(')')
                return node
            if t[0] == 'IDENT':
                func = t[1]
                expect_punct('(')
                if peek() != ('P', ')'):
                    first = parse_function_arg(func)
                    args = [first]
                    if func == 'IF':
                        # parse IF specially after left
                        left = args[0]
                        op_tok = next_()
                        if op_tok[0] != 'OP' or op_tok[1] not in ('=', '<>', '<', '<=', '>', '>='):
                            raise ValueError('expected if op')
                        op = op_tok[1]
                        right = parse_expr()
                        expect_punct(',')
                        then = parse_expr()
                        expect_punct(',')
                        els = parse_expr()
                        expect_punct(')')
                        return ('if', op, left, right, then, els)
                    else:
                        while peek() == ('P', ','):
                            next_()
                            args.append(parse_function_arg(func))
                        expect_punct(')')
                        return ('call', func, args)
                else:
                    expect_punct(')')
                    if func == 'IF':
                        raise ValueError('bad IF')
                    raise ValueError('bad call')
            raise ValueError('bad atom')

        def parse_function_arg(func):
            if func in ('SUM', 'MIN', 'MAX'):
                t = peek(0)
                if t[0] == 'CELL' and peek(1)[0] == 'P' and peek(1)[1] == ':' and peek(2)[0] == 'CELL':
                    a = next_()[1]
                    next_()
                    b = next_()[1]
                    return ('range', a, b)
            return parse_expr()

        node = parse_expr()
        if peek() != ('EOF', ''):
            raise ValueError('trailing tokens')
        return node

    formulas: dict[str, Any] = {}

    def col_index(name: str) -> int:
        return ord(name[0]) - ord('A')

    def row_index(name: str) -> int:
        return int(name[1:])

    def cell_name(col: int, row: int) -> str:
        return chr(ord('A') + col) + str(row)

    def range_iter(a: str, b: str):
        c1, c2 = col_index(a), col_index(b)
        r1, r2 = row_index(a), row_index(b)
        for r in range(min(r1, r2), max(r1, r2) + 1):
            for c in range(min(c1, c2), max(c1, c2) + 1):
                yield cell_name(c, r)

    def eval_node(node):
        kind = node[0]
        if kind == 'int':
            return node[1]
        if kind == 'cell':
            return eval_cell(node[1])
        if kind == 'neg':
            v = eval_node(node[1])
            if is_error(v):
                return v
            return -v
        if kind == 'bin':
            _, op, left, right = node
            lv = eval_node(left)
            if is_error(lv):
                return lv
            rv = eval_node(right)
            if is_error(rv):
                return rv
            if op == '+':
                return lv + rv
            if op == '-':
                return lv - rv
            if op == '*':
                return lv * rv
            if op == '/':
                if rv == 0:
                    return '#DIV/0!'
                q = abs(lv) // abs(rv)
                if (lv < 0) != (rv < 0):
                    q = -q
                return q
        if kind == 'if':
            _, op, left, right, then, els = node
            lv = eval_node(left)
            if is_error(lv):
                return lv
            rv = eval_node(right)
            if is_error(rv):
                return rv
            if op == '=':
                cond = lv == rv
            elif op == '<>':
                cond = lv != rv
            elif op == '<':
                cond = lv < rv
            elif op == '<=':
                cond = lv <= rv
            elif op == '>':
                cond = lv > rv
            elif op == '>=':
                cond = lv >= rv
            else:
                raise ValueError('bad op')
            return eval_node(then if cond else els)
        if kind == 'call':
            _, func, args = node
            values = []
            for arg in args:
                if arg[0] == 'range':
                    for nm in range_iter(arg[1], arg[2]):
                        if nm not in cells:
                            continue
                        v = eval_cell(nm)
                        if is_error(v):
                            return v
                        values.append(v)
                else:
                    v = eval_node(arg)
                    if is_error(v):
                        return v
                    values.append(v)
            if func == 'SUM':
                return sum(values)
            if func == 'MIN':
                return min(values) if values else 0
            if func == 'MAX':
                return max(values) if values else 0
        raise ValueError('bad node')

    def eval_cell(nm: str):
        if nm not in cells:
            return 0
        if nm in memo:
            return memo[nm]
        if nm in in_progress:
            return '#CYCLE!'
        raw = cells[nm]
        if re.fullmatch(r'-?[0-9]+', raw):
            v = int(raw)
            memo[nm] = v
            return v
        if nm in in_progress:
            return '#CYCLE!'
        in_progress.add(nm)
        try:
            if nm not in formulas:
                toks = tokenize(raw[1:])
                formulas[nm] = parse(toks)
            v = eval_node(formulas[nm])
        finally:
            in_progress.remove(nm)
        memo[nm] = v
        return v

    return {nm: eval_cell(nm) for nm in cells}
