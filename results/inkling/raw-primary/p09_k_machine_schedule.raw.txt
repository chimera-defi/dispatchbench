def k_machine_schedule(jobs: list[list[int]], k: int) -> int:
    if not jobs:
        return 0
    times = sorted({t for s, e, p in jobs for t in (s, e)})
    n_times = len(times)
    idx = {t: i for i, t in enumerate(times)}
    N = n_times
    source, sink = 0, N - 1
    adj = [[] for _ in range(N)]

    def add_edge(u, v, cap, cost):
        adj[u].append([v, cap, cost, len(adj[v])])
        adj[v].append([u, 0, -cost, len(adj[u]) - 1])

    for i in range(N - 1):
        add_edge(i, i + 1, k, 0)
    for s, e, p in jobs:
        add_edge(idx[s], idx[e], 1, -p)

    flow = 0
    total_cost = 0
    while flow < k:
        INF = float('inf')
        dist = [INF] * N
        dist[source] = 0
        prev_u = [-1] * N
        prev_ei = [-1] * N
        for _ in range(N - 1):
            updated = False
            for u in range(N):
                du = dist[u]
                if du == INF:
                    continue
                for ei, e in enumerate(adj[u]):
                    if e[1] > 0 and du + e[2] < dist[e[0]]:
                        dist[e[0]] = du + e[2]
                        prev_u[e[0]] = u
                        prev_ei[e[0]] = ei
                        updated = True
            if not updated:
                break
        if dist[sink] == INF:
            break
        bf = k - flow
        v = sink
        path = []
        while v != source:
            u = prev_u[v]
            ei = prev_ei[v]
            path.append((u, ei))
            c = adj[u][ei][1]
            if c < bf:
                bf = c
            v = u
        v = sink
        for u, ei in path:
            e = adj[u][ei]
            v = e[0]
            rev = adj[v][e[3]]
            e[1] -= bf
            rev[1] += bf
        total_cost += bf * dist[sink]
        flow += bf

    return -total_cost
