    k=len(A)
    M=[row[:] + [b[i]] for i,row in enumerate(A)]  # rows length k+1
    for col in range(k):
        pivot=None
        for i in range(col,k):
            if M[i][col] != 0:
                pivot=i; break
        if pivot is None: raise ZeroDivisionError
        if pivot != col:
            M[col],M[pivot]=M[pivot],M[col]
        pv=M[col][col]
        # normalize pivot row
        M[col]=[x/pv for x in M[col]]
        for i in range(k):
            if i==col: continue
            factor=M[i][col]
            if factor==0: continue
            M[i]=[x - factor*y for x,y in zip(M[i], M[col])]
    return [M[i][k] for i in range(k)]
