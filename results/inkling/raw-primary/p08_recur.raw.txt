from __future__ import annotations

import calendar
from datetime import date, timedelta


def recur(start: str, rule: str, count: int) -> list[str]:
    start_date = date.fromisoformat(start)
    parts = rule.split()
    kind = parts[0]
    weekday_index = {"MO": 0, "TU": 1, "WE": 2, "TH": 3, "FR": 4, "SA": 5, "SU": 6}

    if kind == "DAILY":
        step = int(parts[1])
        out = []
        d = start_date
        for _ in range(count):
            out.append(d.isoformat())
            d = d + timedelta(days=step)
        return out

    if kind == "WEEKLY":
        k = int(parts[1])
        idxs = sorted({weekday_index[code] for code in parts[2].split(",")})
        week0_monday = start_date - timedelta(days=start_date.weekday())
        out = []
        j = 0
        while len(out) < count:
            week_start = week0_monday + timedelta(weeks=k * j)
            for idx in idxs:
                d = week_start + timedelta(days=idx)
                if d >= start_date:
                    out.append(d.isoformat())
                    if len(out) == count:
                        return out
            j += 1
        return out

    if kind == "MONTHLY":
        k = int(parts[1])
        if len(parts) == 3:
            day_spec = int(parts[2])
            out = []
            j = 0
            while len(out) < count:
                month_index = start_date.year * 12 + (start_date.month - 1) + k * j
                year, remainder = divmod(month_index, 12)
                month = remainder + 1
                last = calendar.monthrange(year, month)[1]
                if day_spec == -1:
                    day = last
                else:
                    if day_spec > last:
                        j += 1
                        continue
                    day = day_spec
                d = date(year, month, day)
                if d >= start_date:
                    out.append(d.isoformat())
                j += 1
            return out

        ordinal = int(parts[2])
        target = weekday_index[parts[3]]
        out = []
        j = 0
        while len(out) < count:
            month_index = start_date.year * 12 + (start_date.month - 1) + k * j
            year, remainder = divmod(month_index, 12)
            month = remainder + 1
            last = calendar.monthrange(year, month)[1]
            selected = None
            if ordinal == -1:
                for day in range(last, 0, -1):
                    candidate = date(year, month, day)
                    if candidate.weekday() == target:
                        selected = candidate
                        break
            else:
                seen = 0
                for day in range(1, last + 1):
                    candidate = date(year, month, day)
                    if candidate.weekday() == target:
                        seen += 1
                        if seen == ordinal:
                            selected = candidate
                            break
            if selected is not None and selected >= start_date:
                out.append(selected.isoformat())
            j += 1
        return out

    if kind == "YEARLY":
        k = int(parts[1])
        month_s, day_s = parts[2].split("-")
        month = int(month_s)
        day = int(day_s)
        out = []
        j = 0
        while len(out) < count:
            year = start_date.year + k * j
            try:
                d = date(year, month, day)
            except ValueError:
                d = None
            if d is not None and d >= start_date:
                out.append(d.isoformat())
            j += 1
        return out

    raise ValueError("Unsupported rule")
