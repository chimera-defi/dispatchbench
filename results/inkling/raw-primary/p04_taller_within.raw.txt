def taller_within(a: list[int], k: int) -> list[int]:
    n = len(a)
    vals = sorted(set(a))
    rank = {v: i for i, v in enumerate(vals)}
    m = len(vals)
    bit = [0] * (m + 1)

    def add(r, delta):
        i = r + 1
        while i <= m:
            bit[i] += delta
            i += i & -i

    def prefix(r):
        s = 0
        i = r + 1
        while i > 0:
            s += bit[i]
            i -= i & -i
        return s

    res = [0] * n
    window = 0
    for i in range(n):
        rem = i - k - 1
        if rem >= 0:
            add(rank[a[rem]], -1)
            window -= 1
        r = rank[a[i]]
        res[i] = window - prefix(r)
        add(r, 1)
        window += 1
    return res
