def overlap_minutes(meetings: list[list[int]], k: int) -> int:
    delta = {}
    for start, end in meetings:
        if start == end:
            continue
        delta[start] = delta.get(start, 0) + 1
        delta[end] = delta.get(end, 0) - 1

    keys = sorted(delta)
    active = 0
    total = 0
    for i, x in enumerate(keys):
        active += delta[x]
        if i + 1 < len(keys):
            y = keys[i + 1]
            if active >= k:
                total += y - x
    return total
