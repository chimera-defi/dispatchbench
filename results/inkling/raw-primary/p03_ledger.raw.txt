def ledger(ops: list[str]) -> tuple[dict[str, int], list[int]]:
    def valid_amount(token: str) -> bool:
        return len(token) > 0 and all('0' <= ch <= '9' for ch in token)

    balances: dict[str, int] = {}
    frozen: set[str] = set()
    undo_stack: list[tuple[str, str, str | None, int]] = []
    rejected: list[int] = []

    for idx, raw in enumerate(ops):
        parts = raw.split()
        if not parts:
            rejected.append(idx)
            continue
        verb = parts[0]

        if verb == "open":
            if len(parts) != 3 or not valid_amount(parts[2]):
                rejected.append(idx)
                continue
            account, amount = parts[1], int(parts[2])
            if account in balances:
                rejected.append(idx)
                continue
            balances[account] = amount
            frozen.discard(account)
            undo_stack.append(("open", account, None, amount))

        elif verb == "deposit":
            if len(parts) != 3 or not valid_amount(parts[2]):
                rejected.append(idx)
                continue
            account, amount = parts[1], int(parts[2])
            if amount <= 0 or account not in balances:
                rejected.append(idx)
                continue
            balances[account] += amount
            undo_stack.append(("deposit", account, None, amount))

        elif verb == "withdraw":
            if len(parts) != 3 or not valid_amount(parts[2]):
                rejected.append(idx)
                continue
            account, amount = parts[1], int(parts[2])
            if amount <= 0 or account not in balances or account in frozen or balances[account] < amount:
                rejected.append(idx)
                continue
            balances[account] -= amount
            undo_stack.append(("withdraw", account, None, amount))

        elif verb == "transfer":
            if len(parts) != 4 or not valid_amount(parts[3]):
                rejected.append(idx)
                continue
            from_account, to_account, amount = parts[1], parts[2], int(parts[3])
            if (
                amount <= 0
                or from_account == to_account
                or from_account not in balances
                or to_account not in balances
                or from_account in frozen
                or balances[from_account] < amount
            ):
                rejected.append(idx)
                continue
            balances[from_account] -= amount
            balances[to_account] += amount
            undo_stack.append(("transfer", from_account, to_account, amount))

        elif verb == "freeze":
            if len(parts) != 2:
                rejected.append(idx)
                continue
            account = parts[1]
            if account not in balances or account in frozen:
                rejected.append(idx)
                continue
            frozen.add(account)
            undo_stack.append(("freeze", account, None, 0))

        elif verb == "unfreeze":
            if len(parts) != 2:
                rejected.append(idx)
                continue
            account = parts[1]
            if account not in balances or account not in frozen:
                rejected.append(idx)
                continue
            frozen.discard(account)
            undo_stack.append(("unfreeze", account, None, 0))

        elif verb == "undo":
            if len(parts) != 1:
                rejected.append(idx)
                continue
            if not undo_stack:
                rejected.append(idx)
                continue
            record = undo_stack.pop()
            verb_rec, account, other, amount = record
            if verb_rec == "open":
                balances.pop(account, None)
                frozen.discard(account)
            elif verb_rec == "deposit":
                balances[account] -= amount
            elif verb_rec == "withdraw":
                balances[account] += amount
            elif verb_rec == "transfer":
                balances[account] += amount
                balances[other] -= amount
            elif verb_rec == "freeze":
                frozen.discard(account)
            elif verb_rec == "unfreeze":
                frozen.add(account)

        else:
            rejected.append(idx)

    return balances, rejected
