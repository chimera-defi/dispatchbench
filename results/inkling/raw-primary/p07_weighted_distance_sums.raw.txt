def weighted_distance_sums(n, edges, w):
    adj = [[] for _ in range(n)]
    for a, b in edges:
        adj[a].append(b)
        adj[b].append(a)
    parent = [-1]*n
    parent[0] = -2
    order = []
    stack = [0]
    while stack:
        v = stack.pop()
        order.append(v)
        for u in adj[v]:
            if u != parent[v]:
                parent[u] = v
                stack.append(u)
    W = sum(w)
    sub = w[:]  # sub[v] = w[v] + sum children sub
    f = [0]*n
    for v in reversed(order):
        s = w[v]
        fv = 0
        for u in adj[v]:
            if parent[u] == v:
                s += sub[u]
                fv += f[u] + sub[u]
        sub[v] = s
        f[v] = fv
    S = [0]*n
    S[0] = f[0]  # up[0]=0
    # top-down pass
    stack = [0]
    while stack:
        v = stack.pop()
        up_v = S[v] - f[v]  # up[v]

                up_u = up_v + f[v] - f[u] - 2*sub[u] + W
                S[u] = f[u] + up_u
                stack.append(u)
    return S

def weighted_distance_sums(n: int, edges: list[list[int]], w: list[int]) -> list[int]:
    adj = [[] for _ in range(n)]
    for a, b in edges:
        adj[a].append(b)
        adj[b].append(a)

    parent = [-1] * n
    parent[0] = -2
    order = []
    stack = [0]
    while stack:
        v = stack.pop()
        order.append(v)
        for u in adj[v]:
            if parent[u] == -1:
                parent[u] = v
                stack.append(u)

    total_w = sum(w)
    sub = [0] * n  # sub[v] = sum of w in subtree of v
    f = [0] * n    # f[v] = sum_{u in subtree(v)} w[u] * dist(u, v)
    for v in reversed(order):
        s = w[v]
        fv = 0
        for u in adj[v]:
            if parent[u] == v:
                s += sub[u]
                fv += f[u] + sub[u]
        sub[v] = s
        f[v] = fv

    res = [0] * n
    res[0] = f[0]  # up-contribution of root is 0
    stack = [0]
    while stack:
        v = stack.pop()
        up_v = res[v] - f[v]
        for u in adj[v]:
            if parent[u] == v:
                up_u = up_v + f[v] - f[u] - 2 * sub[u] + total_w
                res[u] = f[u] + up_u
                stack.append(u)

    return res
