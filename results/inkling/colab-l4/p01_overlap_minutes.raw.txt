def overlap_minutes(meetings: list[list[int]], k: int) -> int:
    events = []
    for start, end in meetings:
        if start < end:
            events.append((start, 1))
            events.append((end, -1))
    events.sort(key=lambda event: event[0])

    active = 0
    total = 0
    i = 0
    n = len(events)

    while i < n:
        t = events[i][0]
        delta = 0
        while i < n and events[i][0] == t:
            delta += events[i][1]
            i += 1
        active += delta
        if active >= k and i < n:
            total += events[i][0] - t

    return total
