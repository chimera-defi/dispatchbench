def avoid_count(n: int, alphabet: str, banned: list[str]) -> int:
    MOD = 1_000_000_007
    chars = sorted(set(alphabet))
    banned = [b for b in set(banned) if b]
    banned_set = set(banned)
    all_prefixes = {""}
    for b in banned:
        for i in range(1, len(b) + 1):
            all_prefixes.add(b[:i])
    states = [p for p in all_prefixes if p not in banned_set]
    states.sort(key=lambda s: (s != "", s))  # "" first
    idx = {s: i for i, s in enumerate(states)}
    m = len(states)
    maxlen = max((len(b) for b in banned), default=0)

    adj = [[0] * m for _ in range(m)]
    for i, p in enumerate(states):
        plen = len(p)
        for c in chars:
            s = p + c
            found = ""
            L = plen + 1 if plen + 1 < maxlen + 1 else maxlen
            # simpler: L from min(len(s), maxlen) down to 0
            L = min(plen + 1, maxlen)
            while L >= 0:
                suffix = s[plen + 1 - L:] if L else ""
                if suffix in all_prefixes:
                    found = suffix
                    break
                L -= 1
            if found in banned_set:
                continue
            adj[i][idx[found]] = (adj[i][idx[found]] + 1) % MOD

    def matmul(A, B):
        res = [[0] * m for _ in range(m)]
        for i in range(m):
            Ai = A[i]
            res_i = res[i]
            for k in range(m):
                aik = Ai[k]
                if aik == 0:
                    continue
                Bk = B[k]
                for j in range(m):
                    bkj = Bk[j]
                    if bkj:
                        res_i[j] = (res_i[j] + aik * bkj) % MOD
        return res

    def matvec(M, v):
        res = [0] * m
        for i in range(m):
            vi = v[i]
            if vi == 0:
                continue
            row = M[i]
            for j in range(m):
                w = row[j]
                if w:
                    res[j] = (res[j] + vi * w) % MOD
        return res

    start = idx[""]
    v = [0] * m
    v[start] = 1
    base = adj
    k = n
    while k:
        if k & 1:
            v = matvec(base, v)
        k >>= 1
        if k:
            base = matmul(base, base)
    return sum(v) % MOD
