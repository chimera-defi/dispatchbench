def evolve(state: str, offsets: list[int], t: int) -> str:
    n = len(state)
    par = {}
    for d in offsets:
        r = d % n
        par[r] = par.get(r, 0) ^ 1
    base = [r for r, p in par.items() if p]
    cur = bytearray(1 if c == '1' else 0 for c in state)
    bit = 0
    while t:
        if t & 1:
            mult = pow(2, bit, n)
            qmap = {}
            for d in base:
                q = (d * mult) % n
                qmap[q] = qmap.get(q, 0) ^ 1
            qs = [q for q, p in qmap.items() if p]
            nxt = bytearray(n)
            for q in qs:
                if q == 0:
                    for i in range(n):
                        nxt[i] ^= cur[i]
                else:
                    for i in range(n - q):
                        nxt[i] ^= cur[i + q]
                    for i in range(n - q, n):
                        nxt[i] ^= cur[i + q - n]
            cur = nxt
        t >>= 1
        bit += 1
    return ''.join('1' if x else '0' for x in cur)
