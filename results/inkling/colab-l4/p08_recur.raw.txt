from datetime import date, timedelta
import calendar

def recur(start: str, rule: str, count: int) -> list[str]:
    sd = date.fromisoformat(start)
    parts = rule.split()
    kind = parts[0]
    result: list[str] = []

    def add_months(y, m, months):
        total = y * 12 + (m - 1) + months
        return total // 12, total % 12 + 1

    if kind == "DAILY":
        step = int(parts[1])
        d = sd
        while len(result) < count:
            result.append(d.isoformat())
            d += timedelta(days=step)
    elif kind == "WEEKLY":
        step = int(parts[1])
        weekdays = ["MO", "TU", "WE", "TH", "FR", "SA", "SU"]
        offs = sorted({weekdays.index(code) for code in parts[2].split(",")})
        monday0 = sd - timedelta(days=sd.weekday())
        week_index = 0
        while len(result) < count:
            monday = monday0 + timedelta(weeks=week_index)
            for off in offs:
                d = monday + timedelta(days=off)
                if d >= sd:
                    result.append(d.isoformat())
                    if len(result) >= count:
                        break
            if len(result) >= count:
                break
            week_index += step
    elif kind == "MONTHLY":
        step = int(parts[1])
        rel = 0
        if len(parts) == 3:
            dnum = int(parts[2])
            while len(result) < count:
                y, m = add_months(sd.year, sd.month, rel * step)
                last_day = calendar.monthrange(y, m)[1]
                cand = None
                if dnum == -1:
                    cand = date(y, m, last_day)
                elif dnum <= last_day:
                    cand = date(y, m, dnum)
                if cand is not None and cand >= sd:
                    result.append(cand.isoformat())
                rel += 1
        else:
            n = int(parts[2])
            wd_code = parts[3]
            weekdays = ["MO", "TU", "WE", "TH", "FR", "SA", "SU"]
            wd_offset = weekdays.index(wd_code)
            while len(result) < count:
                y, m = add_months(sd.year, sd.month, rel * step)
                last_day = calendar.monthrange(y, m)[1]
                dates = [date(y, m, day) for day in range(1, last_day + 1)
                         if date(y, m, day).weekday() == wd_offset]
                cand = None
                if n == -1:
                    if dates:
                        cand = dates[-1]
                elif n - 1 < len(dates):
                    cand = dates[n - 1]
                if cand is not None and cand >= sd:
                    result.append(cand.isoformat())
                rel += 1
    elif kind == "YEARLY":
        step = int(parts[1])
        mm_dd = parts[2]
        mm, dd = mm_dd.split("-")
        month = int(mm)
        day = int(dd)
        rel = 0
        while len(result) < count:
            y = sd.year + rel * step
            try:
                cand = date(y, month, day)
            except ValueError:
                cand = None
            if cand is not None and cand >= sd:
                result.append(cand.isoformat())
            rel += 1
    return result
