def ledger(ops: list[str]) -> tuple[dict[str, int], list[int]]:
    balances: dict[str, int] = {}
    frozen: set[str] = set()
    stack: list[tuple] = []
    rejected: list[int] = []

    def parse_amount(token: str):
        if not token:
            return None
        for ch in token:
            if ch < '0' or ch > '9':
                return None
        return int(token)

    for i, op in enumerate(ops):
        tokens = op.split()
        if not tokens:
            rejected.append(i)
            continue

        verb = tokens[0]

        if verb == "open":
            if len(tokens) != 3:
                rejected.append(i)
                continue
            a = tokens[1]
            n = parse_amount(tokens[2])
            if n is None:
                rejected.append(i)
                continue
            if a in balances:
                rejected.append(i)
                continue
            balances[a] = n
            frozen.discard(a)
            stack.append(("open", a))
        elif verb == "deposit":
            if len(tokens) != 3:
                rejected.append(i)
                continue
            a = tokens[1]
            n = parse_amount(tokens[2])
            if n is None or n <= 0 or a not in balances:
                rejected.append(i)
                continue
            balances[a] += n
            stack.append(("deposit", a, n))
        elif verb == "withdraw":
            if len(tokens) != 3:
                rejected.append(i)
                continue
            a = tokens[1]
            n = parse_amount(tokens[2])
            if n is None or n <= 0 or a not in balances or a in frozen or balances[a] < n:
                rejected.append(i)
                continue
            balances[a] -= n
            stack.append(("withdraw", a, n))
        elif verb == "transfer":
            if len(tokens) != 4:
                rejected.append(i)
                continue
            a, b = tokens[1], tokens[2]
            n = parse_amount(tokens[3])
            if (
                n is None
                or n <= 0
                or a == b
                or a not in balances
                or b not in balances
                or a in frozen
                or balances[a] < n
            ):
                rejected.append(i)
                continue
            balances[a] -= n
            balances[b] += n
            stack.append(("transfer", a, b, n))
        elif verb == "freeze":
            if len(tokens) != 2:
                rejected.append(i)
                continue
            a = tokens[1]
            if a not in balances or a in frozen:
                rejected.append(i)
                continue
            frozen.add(a)
            stack.append(("freeze", a))
        elif verb == "unfreeze":
            if len(tokens) != 2:
                rejected.append(i)
                continue
            a = tokens[1]
            if a not in balances or a not in frozen:
                rejected.append(i)
                continue
            frozen.remove(a)
            stack.append(("unfreeze", a))
        elif verb == "undo":
            if len(tokens) != 1:
                rejected.append(i)
                continue
            if not stack:
                rejected.append(i)
                continue
            rec = stack.pop()
            kind = rec[0]
            if kind == "open":
                a = rec[1]
                balances.pop(a, None)
                frozen.discard(a)
            elif kind == "deposit":
                balances[rec[1]] -= rec[2]
            elif kind == "withdraw":
                balances[rec[1]] += rec[2]
            elif kind == "transfer":
                balances[rec[1]] += rec[3]
                balances[rec[2]] -= rec[3]
            elif kind == "freeze":
                frozen.discard(rec[1])
            elif kind == "unfreeze":
                frozen.add(rec[1])
        else:
            rejected.append(i)

    return balances, rejected
