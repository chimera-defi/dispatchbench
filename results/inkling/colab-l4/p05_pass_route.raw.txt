import heapq

def pass_route(n: int, edges: list[list[int]], stations: list[list[int]], prices: list[int]) -> int:
    m = len(prices)
    masks = 1 << m
    adj = [[] for _ in range(n)]
    for u, v, cost, zone in edges:
        adj[u].append((v, cost, zone))
        adj[v].append((u, cost, zone))

    buy_at = [[] for _ in range(n)]
    for stop, zone in stations:
        if 0 <= zone < m:
            buy_at[stop].append(zone)

    start, target = 0, n - 1
    if start == target:
        return 0

    INF = 10 ** 30
    dist = [[INF] * masks for _ in range(n)]
    dist[start][0] = 0
    pq = [(0, start, 0)]

    while pq:
        d, u, mask = heapq.heappop(pq)
        if d != dist[u][mask]:
            continue
        if u == target:
            return d

        for v, cost, zone in adj[u]:
            eff = 0 if 0 <= zone < m and (mask >> zone) & 1 else cost
            nd = d + eff
            if nd < dist[v][mask]:
                dist[v][mask] = nd
                heapq.heappush(pq, (nd, v, mask))

        for zone in buy_at[u]:
            bit = 1 << zone
            if not (mask & bit):
                nmask = mask | bit
                nd = d + prices[zone]
                if nd < dist[u][nmask]:
                    dist[u][nmask] = nd
                    heapq.heappush(pq, (nd, u, nmask))

    return -1
