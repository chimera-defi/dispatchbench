def k_machine_schedule(jobs: list[list[int]], k: int) -> int:
    if not jobs:
        return 0

    times = sorted({job[0] for job in jobs} | {job[1] for job in jobs})
    pos = {t: i for i, t in enumerate(times)}
    n = len(times)

    graph = []

    def add_edge(u: int, v: int, cap: int, cost: int) -> None:
        graph.append([u, v, cap, cost])
        graph.append([v, u, 0, -cost])

    for i in range(n - 1):
        add_edge(i, i + 1, k, 0)
    for s, e, pay in jobs:
        add_edge(pos[s], pos[e], 1, -pay)

    source, sink = 0, n - 1
    flow = 0
    cost = 0

    while flow < k:
        dist = [float('inf')] * n
        parent = [None] * n
        dist[source] = 0
        for _ in range(n - 1):
            updated = False
            for idx, (u, v, cap, edge_cost) in enumerate(graph):
                if cap > 0 and dist[u] + edge_cost < dist[v]:
                    dist[v] = dist[u] + edge_cost
                    parent[v] = (u, idx)
                    updated = True
            if not updated:
                break

        if dist[sink] == float('inf'):
            break

        v = sink
        while v != source:
            u, idx = parent[v]
            graph[idx][2] -= 1
            graph[idx ^ 1][2] += 1
            v = u

        flow += 1
        cost += dist[sink]

    return -cost
