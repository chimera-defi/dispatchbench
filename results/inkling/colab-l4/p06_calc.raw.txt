def calc(expr: str) -> int | str:
    class _Error(Exception):
        pass

    try:
        tokens = []
        i = 0
        n = len(expr)
        while i < n:
            c = expr[i]
            if c.isspace():
                i += 1
                continue
            if '0' <= c <= '9':
                start = i
                while i < n and '0' <= expr[i] <= '9':
                    i += 1
                tokens.append(('num', int(expr[start:i])))
                continue
            if c in '+-*/%^()':
                tokens.append((c, c))
                i += 1
                continue
            raise _Error()

        pos = 0

        def peek():
            nonlocal pos
            return tokens[pos][0] if pos < len(tokens) else None

        def advance():
            nonlocal pos
            tok = tokens[pos]
            pos += 1
            return tok

        def trunc_div(a, b):
            q = abs(a) // abs(b)
            if (a < 0) != (b < 0):
                q = -q
            return q

        def parse_primary():
            nonlocal pos
            if peek() == '(':
                advance()
                v = parse_additive()
                if peek() != ')':
                    raise _Error()
                advance()
                return v
            if peek() == 'num':
                return advance()[1]
            raise _Error()

        def parse_power():
            nonlocal pos
            base = parse_primary()
            if peek() == '^':
                advance()
                exp = parse_unary()
                if exp < 0:
                    raise _Error()
                return base ** exp
            return base

        def parse_unary():
            nonlocal pos
            if peek() == '-':
                advance()
                return -parse_unary()
            return parse_power()

        def parse_multiplicative():
            nonlocal pos
            v = parse_unary()
            while peek() in ('*', '/', '%'):
                op = advance()[0]
                rhs = parse_unary()
                if op == '*':
                    v = v * rhs
                elif op == '/':
                    if rhs == 0:
                        raise _Error()
                    v = trunc_div(v, rhs)
                else:
                    if rhs == 0:
                        raise _Error()
                    v = v - rhs * trunc_div(v, rhs)
            return v

        def parse_additive():
            nonlocal pos
            v = parse_multiplicative()
            while peek() in ('+', '-'):
                op = advance()[0]
                rhs = parse_multiplicative()
                if op == '+':
                    v = v + rhs
                else:
                    v = v - rhs
            return v

        result = parse_additive()
        if pos != len(tokens):
            raise _Error()
        return result
    except _Error:
        return "error"
