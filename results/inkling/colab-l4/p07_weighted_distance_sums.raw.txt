def weighted_distance_sums(n: int, edges: list[list[int]], w: list[int]) -> list[int]:
    adj = [[] for _ in range(n)]
    for a, b in edges:
        adj[a].append(b)
        adj[b].append(a)

    parent = [-1] * n
    visited = [False] * n
    visited[0] = True
    order = []
    stack = [0]
    while stack:
        v = stack.pop()
        order.append(v)
        for u in adj[v]:
            if not visited[u]:
                visited[u] = True
                parent[u] = v
                stack.append(u)

    total = sum(w)
    W = [0] * n
    D = [0] * n
    for v in reversed(order):
        W[v] = w[v]
        D[v] = 0
        for u in adj[v]:
            if u == parent[v]:
                continue
            W[v] += W[u]
            D[v] += D[u] + W[u]

    S = [0] * n
    S[0] = D[0]
    for v in order:
        for u in adj[v]:
            if u == parent[v]:
                continue
            S[u] = S[v] + total - 2 * W[u]
    return S
